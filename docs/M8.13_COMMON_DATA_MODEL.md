# M8.13 Common Data Model

## Proposed Common KPI Types

```typescript
// ─── Progress Types ──────────────────────────────────────────────────────────

/** Duration-weighted progress aggregation */
export interface ProgressMetrics {
  /** Duration-weighted progress 0-100 */
  weighted_progress: number;
  /** Total activities (excluding cancelled) */
  total_activities: number;
  /** Activities with progress_percent = 100 or status = 'completed' */
  completed_activities: number;
  /** Activities with progress > 0 and < 100 */
  in_progress_activities: number;
  /** Activities with progress = 0 or null */
  not_started_activities: number;
  /** Total duration hours (denominator for weighting) */
  total_duration_hours: number;
  /** Balance: remaining work duration = total_duration × (1 - weighted_progress/100) */
  balance_duration_hours: number;
}

/** Progress broken down by a grouping dimension */
export interface DimensionProgress {
  /** Grouping key (discipline name, contractor name, unit code, etc.) */
  key: string;
  /** Display label */
  label: string;
  /** Progress metrics for this group */
  metrics: ProgressMetrics;
}

/** Planned progress at a data date (uses baseline or planned dates) */
export interface PlannedProgressMetrics {
  /** Duration-weighted planned % at data date */
  planned_progress: number;
  /** Variance: actual - planned */
  variance: number;
  /** Physical SPI: actual / planned */
  physical_spi: number;
}

/** Full event progress payload */
export interface EventProgressPayload {
  event_id: string;
  organization_id: string;
  data_date: string;  // ISO date
  /** Overall duration-weighted progress */
  overall: ProgressMetrics;
  /** Planned progress and variance */
  planned: PlannedProgressMetrics;
  /** Breakdown by discipline */
  by_discipline: DimensionProgress[];
  /** Breakdown by contractor */
  by_contractor: DimensionProgress[];
  /** Breakdown by unit */
  by_unit: DimensionProgress[];
  /** Breakdown by equipment type (if classified) */
  by_equipment_type: DimensionProgress[];
  /** Scope change impact */
  scope_impact: {
    original_scope_progress: number;
    total_scope_progress: number;
    activities_added_by_scope_change: number;
    progress_impact_percent: number;
  };
}

// ─── Identical Activity Types ────────────────────────────────────────────────

/** Matrix cell: one equipment type × one standard activity type */
export interface IdenticalActivityCell {
  equipment_type_id: string;
  equipment_type_name: string;
  standard_activity_type_id: string;
  standard_activity_name: string;
  standard_activity_code: string;
  total: number;
  completed: number;
  in_progress: number;
  not_started: number;
  balance: number;
  completion_percent: number;
}

/** Full identical activity matrix */
export interface IdenticalActivityMatrix {
  event_id: string;
  /** Total classified activities */
  classified_count: number;
  /** Total unclassified activities */
  unclassified_count: number;
  /** Classification coverage percentage */
  classification_coverage: number;
  /** Matrix rows (one per equipment type × standard activity type combination) */
  cells: IdenticalActivityCell[];
  /** Summary by equipment type */
  by_equipment_type: {
    equipment_type_id: string;
    equipment_type_name: string;
    total: number;
    completed: number;
    completion_percent: number;
  }[];
  /** Summary by standard activity type */
  by_activity_type: {
    standard_activity_type_id: string;
    standard_activity_name: string;
    total: number;
    completed: number;
    completion_percent: number;
  }[];
}

// ─── Executive KPI Types ─────────────────────────────────────────────────────

/** Cross-functional executive KPIs */
export interface ExecutiveKPIPayload {
  event_id: string;
  organization_id: string;
  data_date: string;
  
  // Progress
  overall_progress: number;          // Duration-weighted (M8.13 authoritative)
  planned_progress: number;          // Expected at data_date
  progress_variance: number;         // actual - planned
  physical_spi: number;              // actual / planned
  
  // Schedule
  schedule_health_index: number;     // From ScheduleHealthService (0-100)
  critical_path_activities: number;  // Count of critical activities
  float_distribution: {
    negative: number;
    zero: number;
    positive_low: number;   // 0 < float <= 5 days
    positive_high: number;  // float > 5 days
  };
  
  // Cost/EVM (from M8.10, NOT recalculated)
  evm_spi: number;                   // Cost-weighted SPI from EvmSnapshotService
  evm_cpi: number;                   // CPI from EvmSnapshotService
  evm_eac: number;                   // Estimate at completion
  evm_bac: number;                   // Budget at completion
  evm_data_fresh: boolean;           // Is EVM snapshot recent (<24h)?
  
  // Safety
  total_manpower_today: number;
  total_manhours_today: number;
  open_safety_incidents: number;
  lti_count: number;
  
  // Materials (from M8.12)
  material_readiness_percent: number;
  material_blocked_activities: number;
  
  // Scope (from M8.11)
  open_scope_changes: number;
  scope_change_impact_percent: number;
  
  // Constraints
  open_constraints: number;
  critical_constraints: number;
  
  // Forecast
  forecast_completion_date: string | null;   // ISO date
  forecast_variance_days: number;            // vs planned end
  
  // Activities
  activities_planned_today: number;
  activities_in_progress: number;
  activities_completed: number;
  activities_delayed: number;
  total_activities: number;
  
  // Balance
  balance_duration_hours: number;
  balance_activities: number;
}
```

## CDM Design Principles

1. **Activity.progress_percent is the single source of truth** for execution progress
2. **Duration-weighted aggregation** is the standard method
3. **Cost-weighted aggregation** is EVM-specific and maintained by M8.10
4. **No duplicated calculation** — every KPI computed in exactly one service
5. **Dimensions are additive** — discipline + contractor + unit all derive from the same activities
6. **Scope change awareness** — original vs total scope tracked by baseline tagging
7. **Cross-functional assembly** — executive KPIs assembled from specialist services, not recalculated
8. **Protected module boundary** — M8.13 READS from specialist services, NEVER modifies them
