```typescript
  request: NextRequest,
  { params, tenantContext }: { params: { eventId: string }; tenantContext: { organizationId: string } }
) => {
  // ...
  const data = await SomeService.method(
    params.eventId,
    tenantContext.organizationId,
    // ...

```

**After (correct):**
```typescript
import { withTenantGuard } from '@/lib/withTenantGuard';

export const GET = withTenantGuard(async (
  req: NextRequest,
  ctx,
  session
) => {
  const orgId = session.user.organization_id;
  const { eventId } = await ctx.params;
  // ...
  const data = await SomeService.method(
    eventId,
    orgId,
    // ...
  );
```
