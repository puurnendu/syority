## New Models

### 1. MaterialSupplyRecord [NEW]

Purpose: Track individual supply/delivery records per material requirement.

```prisma
model MaterialSupplyRecord {
  id                  String    @id @default(uuid()) @db.Uuid
  organization_id     String    @db.Uuid
  material_line_id    String    @db.Uuid    // FK to workpack_material_lines
  supplier_name       String?
  po_number           String?
  po_line_number      String?
  quantity_ordered    Float     @default(0)
  quantity_received   Float     @default(0)
  expected_delivery   DateTime? @db.Date    // ETA
  actual_delivery     DateTime? @db.Date
  delivery_status     String    @default("pending")
  // Status: pending, in_transit, partial, received, delayed, cancelled
  unit_cost           Float?
  notes               String?
  created_at          DateTime  @default(now())
  updated_at          DateTime  @updatedAt

  materialLine        workpack_material_lines @relation(fields: [material_line_id], references: [id])
  organization        Organization @relation(fields: [organization_id], references: [id])

  @@index([material_line_id])
  @@index([organization_id])
  @@map("material_supply_records")
}
```

### 2. MaterialConstraint [NEW]

Purpose: Computed constraint record linking material readiness to activity scheduling.

```prisma
model MaterialConstraint {
  id                    String    @id @default(uuid()) @db.Uuid
  organization_id       String    @db.Uuid
  event_id              String    @db.Uuid
  activity_id           String    @db.Uuid
  workpack_id           String?   @db.Uuid
  material_line_id      String?   @db.Uuid
  constraint_type       String    @default("material_eta")
  // Types: material_eta, material_shortage, material_critical
  constraint_date       DateTime? @db.Date    // Effective constraint date
  readiness_status      String    @default("not_ready")
  // Status: ready, partial, not_ready, at_risk, blocked
  readiness_percent     Float     @default(0)
  quantity_required     Float     @default(0)
  quantity_available    Float     @default(0)
  quantity_on_order     Float     @default(0)
  earliest_eta          DateTime? @db.Date
  is_binding            Boolean   @default(false)  // Is this the binding constraint?
  is_critical_material  Boolean   @default(false)
  impact_days           Float     @default(0)       // Days of schedule impact
  last_calculated_at    DateTime  @default(now())
  created_at            DateTime  @default(now())
  updated_at            DateTime  @updatedAt

  activity              Activity     @relation(fields: [activity_id], references: [id])
  event                 Event        @relation(fields: [event_id], references: [id])
  organization          Organization @relation(fields: [organization_id], references: [id])

  @@unique([activity_id, material_line_id])
  @@index([event_id, readiness_status])
  @@index([activity_id])
  @@index([organization_id])
  @@map("material_constraints")
}
```
